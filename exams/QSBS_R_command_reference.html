QSBS — R command reference for the exam

1 How to use this reference

This document lists all R commands that were used in the QSBS practicals, organised by statistical topic rather than by practical. You may consult it during the exam. It is a look-up document: you are not expected to remember the commands, but you are expected to know which command solves a given problem, what its arguments mean and how to read its output.

Conventions: x, y are numeric vectors; f, g are factors (categorical variables); d is a data frame; fm is a fitted model. Names such as mu, sigma, n, p, l are numbers you supply. Anything after # is a comment.

5 Probability distributions

Every distribution has four functions: d (density / probability), p (cumulative probability \(P(X \le q)\)), q (quantile, the inverse of p) and r (random numbers).

# Discrete
dbinom(k, size = n, prob = p); pbinom(q, size = n, prob = p); qbinom(p, size = n, prob = p); rbinom(m, size = n, prob = p)
dpois(k, lambda = l);          ppois(q, lambda = l);          qpois(p, lambda = l);          rpois(m, lambda = l)
dnbinom(k, size = theta, mu = m)

# Continuous
dnorm(x, mean = mu, sd = sigma); pnorm(q, mean = mu, sd = sigma); qnorm(p, mean = mu, sd = sigma); rnorm(m, mean = mu, sd = sigma)
dunif(x, min = a, max = b)
dlnorm(x, meanlog = mu, sdlog = sigma); plnorm(); qlnorm()
dgamma(x, shape = k, scale = theta)
dweibull(x, shape = k, scale = lambda)
dlogis(x, location = mu, scale = s)
plogis(eta)                              # inverse logit: e^eta / (1 + e^eta)
qlogis(p)                                # logit: log(p / (1 - p))

# Upper tail and intervals
pnorm(q, mu, sigma, lower.tail = FALSE)  # P(X > q)
ppois(25, l) - ppois(19, l)              # P(20 <= X <= 25) for a discrete variable
sum(dbinom(20:25, size = 30, prob = 0.8))   # the same by summation
integrate(dnorm, lower = -Inf, upper = 2)   # area under a density

# Plotting distributions
curve(dnorm(x, mean = 0, sd = 1), from = -4, to = 4)
curve(dnorm(x, mean = 0, sd = 2), add = TRUE, col = "grey50")
k <- 0:20; plot(k, dpois(k, lambda = 4), type = "h", lwd = 2); points(k, dpois(k, 4), pch = 19)
abline(v = 2, col = "grey"); abline(h = 0.5, lty = 2); segments(x0, y0, x1, y1)
polygon(c(xs, rev(xs)), c(ys, rep(0, length(ys))), col = "grey80")   # shade an area

# Comparing a distribution with data
hist(x, probability = TRUE); curve(dnorm(x, mean(x), sd(x)), add = TRUE)
mean(x <= 2.5)                           # observed proportion vs pnorm(2.5, ...)
quantile(x, 0.90)                        # observed quantile vs qnorm(0.90, ...)

6 Likelihood, fitting distributions and comparing fits

# Likelihood by hand
dbinom(60, size = 100, prob = 0.5)                 # likelihood of one observation
prod(dnorm(x, mu, sigma))                          # likelihood of a data set (product!)
sum(dnorm(x, mu, sigma, log = TRUE))               # log-likelihood (sum)
pgrid <- seq(0, 1, length = 200); ll <- dbinom(60, 100, pgrid, log = TRUE); plot(pgrid, ll, type = "l")
pgrid[which.max(ll)]                               # maximum likelihood estimate on a grid

# Maximum likelihood fit of a distribution (MASS)
fit <- fitdistr(x, "normal")       # also "lognormal", "gamma", "weibull", "logistic", "Poisson", "negative binomial"
fit$estimate; fit$estimate[1]; fit$sd              # estimates and their standard errors
logLik(fit); AIC(fit); BIC(fit)                    # AIC = -2 logL + 2p ; BIC = -2 logL + log(n) p ; lower is better

# Goodness of fit
ks.test(x, "pnorm", fit$estimate[1], fit$estimate[2])   # Kolmogorov-Smirnov test
shapiro.test(x)                                          # test of normality
qqnorm(x); qqline(x)
hist(x, probability = TRUE); curve(dgamma(x, fit$estimate[1], fit$estimate[2]), add = TRUE)
plot(ecdf(x), verticals = TRUE, pch = ""); curve(pnorm(x, fit$estimate[1], fit$estimate[2]), add = TRUE, col = "red")

# Maximum likelihood by hand with optim()
llfun <- function(parms) {
  mu <- exp(parms[1] + parms[2] * d$x)          # model on the link scale, back-transformed
  sum(dpois(d$y, lambda = mu, log = TRUE))
}
op <- optim(par = c(1, 0.1), fn = llfun, control = list(fnscale = -1), method = "BFGS", hessian = TRUE)
op$par; op$value; op$convergence
sqrt(diag(solve(-op$hessian)))                   # standard errors
-2 * op$value + 2 * length(op$par)               # AIC by hand
curve(p["a"] * x / (p["b"] + x), add = TRUE)     # draw a hand-fitted curve

7 Generalised linear models: fitting and reading the output

7.1 Fitting

fm <- glm(y ~ x, data = d, family = gaussian)                     # normal errors, identity link (= lm)
fm <- lm(y ~ x, data = d)                                         # the same, classical syntax
fm <- glm(y ~ x, data = d, family = gaussian(link = "log"))       # normal errors, log link
fm <- glm(y ~ x, data = d, family = poisson)                      # counts, log link
fm <- glm(cbind(succ, fail) ~ x, data = d, family = binomial)     # proportions, logit link
fm <- glm(y01 ~ x, data = d, family = binomial)                   # 0/1 data, logit link
fm <- glm(y ~ x, data = d, family = quasipoisson)                 # over/underdispersed counts
fm <- glm(cbind(succ, fail) ~ x, data = d, family = quasibinomial)
fm <- glm.nb(y ~ x, data = d)                                     # negative binomial (MASS)
fm <- glm(y ~ f, data = d, family = poisson)                      # factor as predictor
fm <- glm(y ~ x + f, data = d, family = poisson)                  # several predictors (additive)
fm <- glm(y ~ x * f, data = d, family = poisson)                  # with interaction (= x + f + x:f)
fm <- glm(y ~ poly(x, 2), data = d, family = poisson)             # quadratic (orthogonal polynomial)
fm <- glm(y ~ x + I(x^2), data = d, family = poisson)             # quadratic, explicit coefficients
fm <- glm(y ~ log(x), data = d, family = poisson)                 # transformed predictor
fm0 <- glm(y ~ 1, data = d, family = poisson)                     # intercept-only (null) model
fm2 <- update(fm, . ~ . + z)                                      # add a term
fm2 <- update(fm, . ~ . - x:f)                                    # drop a term
fm2 <- update(fm, data = d[-c(4, 9), ])                           # refit without rows 4 and 9
fm2 <- update(fm, family = quasipoisson)                          # refit with another family

7.2 What is inside the result

summary(fm)                              # coefficients, SE, tests, deviances, dispersion, AIC
summary(fm)$coefficients                 # the coefficient table only
summary(fm)$dispersion                   # dispersion parameter (estimated for gaussian / quasi families)
coef(fm); confint(fm)                    # estimates; confidence intervals
fitted(fm); residuals(fm)                # fitted values; deviance residuals
deviance(fm); fm$null.deviance; df.residual(fm)
logLik(fm); AIC(fm); BIC(fm)
fm$theta                                 # dispersion parameter theta of a glm.nb fit
model.matrix(~ f, data = d)              # how R codes a factor into dummy variables
levels(d$f); d$f <- relevel(d$f, ref = "other level")

# Deviance explained
(fm$null.deviance - deviance(fm)) / fm$null.deviance

7.3 Back-transforming coefficients and predicting

exp(coef(fm))                                   # log link: multiplicative effects
plogis(coef(fm)["(Intercept)"])                 # logit link: probability at x = 0
predict(fm, newdata = data.frame(x = 3))                     # linear predictor (link scale)
predict(fm, newdata = data.frame(x = 3), type = "response")  # on the scale of the data
predict(fm, newdata = data.frame(x = 3, f = "level"), type = "response")
dose.p(fm, p = 0.5)                             # x at which p = 0.5 in a binomial GLM (MASS)

7.4 Drawing a fitted curve with a 95 % confidence band

xnew <- seq(min(d$x), max(d$x), length = 100)
pr <- predict(fm, newdata = data.frame(x = xnew), se.fit = TRUE)    # on the link scale
plot(y ~ x, data = d)
lines(xnew, exp(pr$fit))                                            # exp() for log link, plogis() for logit link
lines(xnew, exp(pr$fit + 2 * pr$se.fit), lty = 2)
lines(xnew, exp(pr$fit - 2 * pr$se.fit), lty = 2)
matlines(xnew, exp(cbind(pr$fit, pr$fit - 2 * pr$se.fit, pr$fit + 2 * pr$se.fit)), lty = c(1, 2, 2), col = "black")
plot(y ~ x, data = d, log = "y")                                    # logarithmic y-axis (= link scale for a log link)
curve(exp(coef(fm)[1] + coef(fm)[2] * x), add = TRUE)               # fitted curve from the coefficients
# one curve per level of a factor
for (lev in levels(d$f)) lines(xnew, predict(fm, newdata = data.frame(x = xnew, f = lev), type = "response"))

8 Regression diagnostics

# 1. Influential points
cd <- cooks.distance(fm)
plot(cd, type = "h"); abline(h = c(1, 4 / nrow(d)), lty = 2)      # thresholds 1 and 4/n
which(cd > 1); which(cd > 4 / nrow(d))
plot(fm, which = 4)                                               # built-in Cook's distance plot
hatvalues(fm); influence.measures(fm)
fm_red <- update(fm, data = d[-i, ]); cbind(coef(fm), coef(fm_red))   # refit without point i

# 2. Dispersion
deviance(fm) / df.residual(fm)                                    # should be about 1 (Poisson, binomial)
sum(residuals(fm, type = "pearson")^2) / df.residual(fm)          # Pearson version (used by quasi families)
update(fm, family = quasipoisson); update(fm, family = quasibinomial)   # SE scaled by sqrt(dispersion)
glm.nb(y ~ x, data = d)                                           # alternative for overdispersed counts

# 3. Residuals
rSD <- rstandard(fm)                     # standardised deviance residuals
eta <- predict(fm, type = "link")        # fitted values on the link scale
plot(eta, rSD); abline(h = 0, lty = 3); lines(lowess(eta, rSD), col = "red")
plot(d$x, rSD); abline(h = 0, lty = 3)   # against each predictor
boxplot(rSD ~ d$f)                       # against a factor
which(abs(rSD) > 2)
residuals(fm, type = "pearson"); residuals(fm, type = "deviance"); rstudent(fm)

# 4. Functional relationship: compare with alternative models
fm2 <- glm(y ~ poly(x, 2), data = d, family = poisson)
anova(fm, fm2, test = "Chisq")           # nested models, fixed dispersion (poisson, binomial)
anova(fm, fm2, test = "F")               # nested models, estimated dispersion (gaussian, quasi)
anova(fnb, fnb2)                         # nested negative binomial models
AIC(fm, fm2, fnb)                        # any models with a true likelihood (not quasi)

9 Multiple regression

# Formulae
y ~ x1 + x2                # additive
y ~ x1 * x2                # with interaction (= x1 + x2 + x1:x2)
y ~ x1 + x2 + x1:x2        # the same, written out
y ~ x * f                  # continuous x categorical: separate slopes per level

# Sequential analysis of deviance / variance of ONE model (order of terms matters)
anova(fm, test = "F")      # gaussian / quasi
anova(fm, test = "Chisq")  # poisson / binomial
anova(fm)                  # lm

# Comparing nested models
anova(fm_small, fm_big, test = "F")
anova(fm_small, fm_big, test = "Chisq")

# Collinearity
cor(d[, c("x1", "x2", "x3")]); pairs(d[, c("x1", "x2", "x3")])
vif(fm)                    # variance inflation factors (car); > 10 is a problem

# Model selection
AIC(fm1, fm2, fm3); BIC(fm1, fm2, fm3)
step(fm)                                   # backward stepwise by AIC
step(fm, k = log(nrow(d)))                 # backward stepwise by BIC
step(fm, trace = 0)                        # without printing the steps
options(na.action = "na.fail"); dredge(fm, rank = "AIC")   # all sub-models (MuMIn)
subset(dredge(fm), delta < 2)

# Visualising two continuous predictors
x1seq <- seq(min(d$x1), max(d$x1), length = 50); x2seq <- seq(min(d$x2), max(d$x2), length = 50)
z <- outer(x1seq, x2seq, function(a, b) predict(fm, newdata = data.frame(x1 = a, x2 = b)))
image(x1seq, x2seq, z); contour(x1seq, x2seq, z, add = TRUE); points(d$x1, d$x2)
